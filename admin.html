function formatStudentDate(createdAt) { try { if (!createdAt) return 'غير محدد'; const d = typeof createdAt.toDate === 'function' ? createdAt.toDate() : new Date(createdAt); if (isNaN(d.getTime())) return 'غير محدد'; return d.toLocaleDateString('ar-SY'); } catch (e) { return 'غير محدد'; } } function getStudentStatusBadge(student) { if (student.approved === true || student.status === 'مقبول') { return `مقبول`; } if (student.status === 'مرفوض') { return `مرفوض`; } return `قيد الانتظار`; } async function loadStudentsManagement() { const tbody = document.getElementById('admin-students-tbody'); if (!tbody) return; tbody.innerHTML = `جاري تحميل الطلاب...`; try { const snap = await db.collection('users').where('role', '==', 'student').get(); if (snap.empty) { tbody.innerHTML = `لا توجد طلبات تسجيل للطلاب حالياً.`; return; } const students = []; snap.forEach(doc => students.push({ id: doc.id, ...doc.data() })); students.sort((a, b) => { const ta = a.createdAt && typeof a.createdAt.toMillis === 'function' ? a.createdAt.toMillis() : 0; const tb = b.createdAt && typeof b.createdAt.toMillis === 'function' ? b.createdAt.toMillis() : 0; return tb - ta; }); let html = ''; for (const st of students) { const selectedGrade = st.allowedGrade || st.gradeRequested || 'الصف السابع'; html += ` ${st.name || '-'} ${st.email || '-'} ${st.gradeRequested || '-'} ${formatStudentDate(st.createdAt)} ${getStudentStatusBadge(st)}
`; } tbody.innerHTML = html; } catch (err) { tbody.innerHTML = `تعذر تحميل قائمة الطلاب: ${err.message}`; } } async function approveStudent(studentId) { const selectEl = document.getElementById(`allowed-grade-${studentId}`); const allowedGrade = selectEl ? selectEl.value : 'الصف السابع'; try { await db.collection('users').doc(studentId).update({ approved: true, status: 'مقبول', allowedGrade: allowedGrade }); await loadStudentsManagement(); alert(`تم قبول الطالب وتحديد الصف المسموح له: ${allowedGrade}`); } catch (err) { alert('حدث خطأ أثناء قبول الطالب: ' + err.message); } } async function rejectStudent(studentId) { try { await db.collection('users').doc(studentId).update({ approved: false, status: 'مرفوض' }); await loadStudentsManagement(); alert('تم رفض حساب الطالب وتعيين approved = false.'); } catch (err) { alert('حدث خطأ أثناء رفض الطالب: ' + err.message); } } async function deleteStudentRecord(studentId) { if (!confirm('هل أنت متأكد من حذف سجل هذا الطالب من قاعدة البيانات؟')) return; try { await db.collection('users').doc(studentId).delete(); await loadStudentsManagement(); alert('تم حذف سجل الطالب بنجاح.'); } catch (err) { alert('حدث خطأ أثناء حذف الطالب: ' + err.message); } }
